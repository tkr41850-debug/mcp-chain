    ```yaml
        tags:
          - 'v*'

    jobs:
      goreleaser:
        name: GoReleaser
        runs-on: ubuntu-latest
        steps:
          - name: Checkout
            uses: actions/checkout@v4
            with:

          - name: Set up Go
            uses: actions/setup-go@v5

          - name: Run GoReleaser
            uses: goreleaser/goreleaser-action@v6
            with:
              distribution: goreleaser
              version: '~> v2'
              args: release --clean
            env:
              GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
    ```
